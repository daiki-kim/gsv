// Package config loads gsv's user configuration.
package config

import (
	"bytes"
	"errors"
	"fmt"
	"io"
	"io/fs"
	"os"
	"path/filepath"
	"strings"

	"gopkg.in/yaml.v3"
)

// SharedRoot is an extra tree root: a shared drive ID or a folder ID.
type SharedRoot struct {
	Name string `yaml:"name"`
	ID   string `yaml:"id"`
}

type Config struct {
	SharedRoots []SharedRoot `yaml:"shared_roots"`
}

// Dir returns $XDG_CONFIG_HOME/gsv, or ~/.config/gsv when XDG_CONFIG_HOME is unset.
func Dir() (string, error) {
	if x := os.Getenv("XDG_CONFIG_HOME"); x != "" {
		return filepath.Join(x, "gsv"), nil
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, ".config", "gsv"), nil
}

// Load reads the YAML config at path. A missing file yields an empty Config.
func Load(path string) (Config, error) {
	b, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return Config{}, nil
	}
	if err != nil {
		return Config{}, err
	}
	var c Config
	dec := yaml.NewDecoder(bytes.NewReader(b))
	dec.KnownFields(true)
	if err := dec.Decode(&c); err != nil && !errors.Is(err, io.EOF) {
		return Config{}, fmt.Errorf("%s: %w", path, err)
	}
	for i, r := range c.SharedRoots {
		if strings.TrimSpace(r.Name) == "" {
			return Config{}, fmt.Errorf("%s: shared_roots[%d]: name is required", path, i)
		}
		if strings.TrimSpace(r.ID) == "" {
			return Config{}, fmt.Errorf("%s: shared_roots[%d]: id is required", path, i)
		}
	}
	return c, nil
}
```

// Package keymap turns vim-style key sequences into pane-independent actions.
package keymap

import "strconv"

type Kind int

const (
	None Kind = iota
	Up
	Down
	Left
	Right
	Top
	Bottom
	LineStart
	LineEnd
	HalfPageDown
	HalfPageUp
	NextTab
	PrevTab
	PaneLeft
	PaneRight
	PaneNext
	ToggleTree
	Open
	Peek
	Reload
	CommandMode
	SearchForward
	SearchBackward
	NextMatch
	PrevMatch
	Cancel
)

// Action is a completed key sequence. Count is at least 1; HasCount reports
// whether the user typed one explicitly.
type Action struct {
	Kind     Kind
	Count    int
	HasCount bool
}

const maxCount = 99999

var single = map[string]Kind{
	"j": Down, "down": Down,
	"k": Up, "up": Up,
	"h": Left, "left": Left,
	"l": Right, "right": Right,
	"G": Bottom, "0": LineStart, "$": LineEnd,
	"ctrl+d": HalfPageDown, "ctrl+u": HalfPageUp,
	"ctrl+n": ToggleTree,
	"enter": Open, "K": Peek, "r": Reload,
	":": CommandMode, "/": SearchForward, "?": SearchBackward,
	"n": NextMatch, "N": PrevMatch,
}

// Parser accumulates counts and multi-key prefixes (g, ctrl+w). The zero value is ready to use.
type Parser struct {
	count   int
	pending string
}

// Feed consumes one key and returns the completed action, if any.
func (p *Parser) Feed(key string) (Action, bool) {
	if key == "esc" {
		p.reset()
		return Action{Kind: Cancel, Count: 1}, true
	}
	switch p.pending {
	case "g":
		switch key {
		case "g":
			return p.emit(Top)
		case "t":
			return p.emit(NextTab)
		case "T":
			return p.emit(PrevTab)
		}
		p.reset()
		return Action{}, false
	case "ctrl+w":
		switch key {
		case "h", "left", "ctrl+h":
			return p.emit(PaneLeft)
		case "l", "right", "ctrl+l":
			return p.emit(PaneRight)
		case "w", "ctrl+w":
			return p.emit(PaneNext)
		}
		p.reset()
		return Action{}, false
	}
	if len(key) == 1 && key[0] >= '0' && key[0] <= '9' && (key != "0" || p.count > 0) {
		if next := p.count*10 + int(key[0]-'0'); next <= maxCount {
			p.count = next
		} else {
			p.count = maxCount
		}
		return Action{}, false
	}
	if key == "g" || key == "ctrl+w" {
		p.pending = key
		return Action{}, false
	}
	if k, ok := single[key]; ok {
		return p.emit(k)
	}
	p.reset()
	return Action{}, false
}

// Pending returns the in-progress sequence, e.g. "5g" or "^W".
func (p *Parser) Pending() string {
	s := ""
	if p.count > 0 {
		s = strconv.Itoa(p.count)
	}
	switch p.pending {
	case "g":
		s += "g"
	case "ctrl+w":
		s += "^W"
	}
	return s
}

func (p *Parser) emit(k Kind) (Action, bool) {
	a := Action{Kind: k, Count: max(p.count, 1), HasCount: p.count > 0}
	p.reset()
	return a, true
}

func (p *Parser) reset() {
	p.count = 0
	p.pending = ""
}
```

// Package retry retries Google API calls on rate limiting and server errors.
package retry

import (
	"context"
	"errors"
	"time"

	"google.golang.org/api/googleapi"
)

// BaseDelay is the wait before the first retry; it doubles on each retry.
var BaseDelay = 500 * time.Millisecond

const MaxRetries = 3

// Do runs fn and retries it up to MaxRetries times while it fails with HTTP 429 or 5xx.
func Do(ctx context.Context, fn func() error) error {
	delay := BaseDelay
	for attempt := 0; ; attempt++ {
		err := fn()
		if err == nil || attempt == MaxRetries || !retryable(err) {
			return err
		}
		select {
		case <-time.After(delay):
		case <-ctx.Done():
			return ctx.Err()
		}
		delay *= 2
	}
}

func retryable(err error) bool {
	var gerr *googleapi.Error
	if errors.As(err, &gerr) {
		return gerr.Code == 429 || gerr.Code >= 500
	}
	return false
}

package gsheets

import (
	"context"
	"fmt"
	"strings"

	"google.golang.org/api/sheets/v4"

	"gsv/internal/retry"
)

type Tab struct {
	ID    int64
	Title string
	Rows  int64
	Cols  int64
}

type Client interface {
	GetTabs(ctx context.Context, spreadsheetID string) (title string, tabs []Tab, err error)
	GetValues(ctx context.Context, spreadsheetID, tabTitle string) ([][]string, error)
}

type APIClient struct {
	svc *sheets.Service
}

func New(svc *sheets.Service) *APIClient { return &APIClient{svc: svc} }

// A1Range quotes a tab title so it can be used as a whole-sheet A1 range.
func A1Range(tab string) string {
	return "'" + strings.ReplaceAll(tab, "'", "''") + "'"
}

func (c *APIClient) GetTabs(ctx context.Context, id string) (string, []Tab, error) {
	var ss *sheets.Spreadsheet
	err := retry.Do(ctx, func() error {
		var err error
		ss, err = c.svc.Spreadsheets.Get(id).
			Fields("properties.title,sheets.properties(sheetId,title,gridProperties)").
			Context(ctx).Do()
		return err
	})
	if err != nil {
		return "", nil, err
	}
	var tabs []Tab
	for _, s := range ss.Sheets {
		p := s.Properties
		if p == nil {
			continue
		}
		t := Tab{ID: p.SheetId, Title: p.Title}
		if g := p.GridProperties; g != nil {
			t.Rows, t.Cols = g.RowCount, g.ColumnCount
		}
		tabs = append(tabs, t)
	}
	title := ""
	if ss.Properties != nil {
		title = ss.Properties.Title
	}
	return title, tabs, nil
}

func (c *APIClient) GetValues(ctx context.Context, id, tab string) ([][]string, error) {
	var vr *sheets.ValueRange
	err := retry.Do(ctx, func() error {
		var err error
		vr, err = c.svc.Spreadsheets.Values.Get(id, A1Range(tab)).
			ValueRenderOption("FORMATTED_VALUE").
			Context(ctx).Do()
		return err
	})
	if err != nil {
		return nil, err
	}
	rows := make([][]string, len(vr.Values))
	for i, row := range vr.Values {
		rows[i] = make([]string, len(row))
		for j, cell := range row {
			if cell != nil {
				rows[i][j] = fmt.Sprint(cell)
			}
		}
	}
	return rows, nil
}

// Package gdrive lists folders and spreadsheets from Google Drive.
package gdrive

import (
	"context"
	"fmt"
	"sort"
	"strings"

	"google.golang.org/api/drive/v3"

	"gsv/internal/retry"
)

type Kind int

const (
	KindFolder Kind = iota
	KindSpreadsheet
)

type Item struct {
	ID   string
	Name string
	Kind Kind
}

type Client interface {
	ListChildren(ctx context.Context, parentID string) ([]Item, error)
	ListSharedWithMe(ctx context.Context) ([]Item, error)
	SearchSpreadsheets(ctx context.Context, name string) ([]Item, error)
}

const (
	mimeFolder   = "application/vnd.google-apps.folder"
	mimeSheet    = "application/vnd.google-apps.spreadsheet"
	mimeShortcut = "application/vnd.google-apps.shortcut"
)

var baseFilter = fmt.Sprintf("trashed = false and (mimeType = '%s' or mimeType = '%s' or mimeType = '%s')",
	mimeFolder, mimeSheet, mimeShortcut)

func escape(s string) string {
	return strings.ReplaceAll(strings.ReplaceAll(s, `\`, `\\`), `'`, `\'`)
}

func ChildrenQuery(parentID string) string {
	return fmt.Sprintf("'%s' in parents and %s", escape(parentID), baseFilter)
}

func SharedQuery() string { return "sharedWithMe = true and " + baseFilter }

func SearchQuery(name string) string {
	return fmt.Sprintf("name contains '%s' and mimeType = '%s' and trashed = false", escape(name), mimeSheet)
}

type APIClient struct {
	svc *drive.Service
}

func New(svc *drive.Service) *APIClient { return &APIClient{svc: svc} }

func (c *APIClient) ListChildren(ctx context.Context, parentID string) ([]Item, error) {
	return c.list(ctx, ChildrenQuery(parentID))
}

func (c *APIClient) ListSharedWithMe(ctx context.Context) ([]Item, error) {
	return c.list(ctx, SharedQuery())
}

func (c *APIClient) SearchSpreadsheets(ctx context.Context, name string) ([]Item, error) {
	return c.list(ctx, SearchQuery(name))
}

func (c *APIClient) list(ctx context.Context, q string) ([]Item, error) {
	var items []Item
	token := ""
	for {
		var fl *drive.FileList
		err := retry.Do(ctx, func() error {
			call := c.svc.Files.List().Q(q).
				OrderBy("folder,name").PageSize(1000).
				SupportsAllDrives(true).IncludeItemsFromAllDrives(true).Corpora("allDrives").
				Fields("nextPageToken, files(id, name, mimeType, shortcutDetails)").
				Context(ctx)
			if token != "" {
				call = call.PageToken(token)
			}
			var err error
			fl, err = call.Do()
			return err
		})
		if err != nil {
			return nil, err
		}
		for _, f := range fl.Files {
			if it, ok := toItem(f); ok {
				items = append(items, it)
			}
		}
		if fl.NextPageToken == "" {
			break
		}
		token = fl.NextPageToken
	}
	// Shortcuts to folders sort among files on the API side; put every folder first.
	sort.SliceStable(items, func(i, j int) bool { return items[i].Kind < items[j].Kind })
	return items, nil
}

func toItem(f *drive.File) (Item, bool) {
	id, mime := f.Id, f.MimeType
	if mime == mimeShortcut {
		if f.ShortcutDetails == nil {
			return Item{}, false
		}
		id, mime = f.ShortcutDetails.TargetId, f.ShortcutDetails.TargetMimeType
	}
	switch mime {
	case mimeFolder:
		return Item{ID: id, Name: f.Name, Kind: KindFolder}, true
	case mimeSheet:
		return Item{ID: id, Name: f.Name, Kind: KindSpreadsheet}, true
	}
	return Item{}, false
}

	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {

func LoadOAuthConfig(credPath string) (*oauth2.Config, error) {
	b, err := os.ReadFile(credPath)

func LoadToken(path string) (*oauth2.Token, error) {
	b, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	var tok oauth2.Token
	if err := json.Unmarshal(b, &tok); err != nil {
		return nil, err
	}
	return &tok, nil
}

// SaveToken writes tok atomically with mode 0600, creating the directory with 0700.
func SaveToken(path string, tok *oauth2.Token) error {
	dir := filepath.Dir(path)
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return err
	}
	f, err := os.CreateTemp(dir, ".token-*")
	if err != nil {
		return err
	}
	defer os.Remove(f.Name())
	if err := f.Chmod(0o600); err != nil {
		f.Close()
		return err
	}
	if err := json.NewEncoder(f).Encode(tok); err != nil {
		f.Close()
		return err
	}
	if err := f.Close(); err != nil {
		return err
	}
	return os.Rename(f.Name(), path)
}

// Authorize runs the loopback flow: it serves /callback on 127.0.0.1, opens the
// consent page and exchanges the returned code (with PKCE) for a token.
func Authorize(ctx context.Context, cfg *oauth2.Config, openURL func(string) error, out io.Writer) (*oauth2.Token, error) {
	ln, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		return nil, err
	}
	c := *cfg
	c.RedirectURL = "http://" + ln.Addr().String() + "/callback"
	state, err := randomState()
	if err != nil {
		ln.Close()
		return nil, err
	}
	verifier := oauth2.GenerateVerifier()
	authURL := c.AuthCodeURL(state, oauth2.AccessTypeOffline, oauth2.S256ChallengeOption(verifier),
		oauth2.SetAuthURLParam("prompt", "consent"))

	codeCh, errCh := make(chan string, 1), make(chan error, 1)
	mux := http.NewServeMux()
	mux.Handle("/callback", callbackHandler(state, codeCh, errCh))
	srv := &http.Server{Handler: mux}
	go srv.Serve(ln)
	defer srv.Close()

	fmt.Fprintf(out, "ブラウザで Google アカウントへのアクセスを許可してください:\n%s\n", authURL)
	if openURL != nil {
		_ = openURL(authURL)
	}
	select {
	case code := <-codeCh:
		return c.Exchange(ctx, code, oauth2.VerifierOption(verifier))
	case err := <-errCh:
		return nil, err
	case <-ctx.Done():
		return nil, ctx.Err()
	}
}

func callbackHandler(state string, codeCh chan<- string, errCh chan<- error) http.Handler {
	fail := func(w http.ResponseWriter, err error) {
		http.Error(w, err.Error(), http.StatusBadRequest)
		select {
		case errCh <- err:
		default:
		}
	}
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		q := r.URL.Query()
		if q.Get("state") != state {
			fail(w, errors.New("oauth: state mismatch"))
			return
		}
		if e := q.Get("error"); e != "" {
			fail(w, errors.New("oauth: "+e))
			return
		}
		code := q.Get("code")
		if code == "" {
			fail(w, errors.New("oauth: missing code"))
			return
		}
		fmt.Fprintln(w, "gsv: 認可が完了しました。ターミナルに戻ってください。")
		select {
		case codeCh <- code:
		default:
		}
	})
}

func randomState() (string, error) {
	b := make([]byte, 24)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b), nil
}

// TokenSource returns a token source backed by tokenPath. It refreshes the stored
// token up front and falls back to the browser flow when the token is missing,
// unreadable or rejected by Google. Refreshed tokens are written back to tokenPath.
func TokenSource(ctx context.Context, cfg *oauth2.Config, tokenPath string, openURL func(string) error, out io.Writer) (oauth2.TokenSource, error) {
	tok, err := LoadToken(tokenPath)
	if err == nil {
		ts := cfg.TokenSource(ctx, tok)
		fresh, err := ts.Token()
		if err == nil {
			if fresh.AccessToken != tok.AccessToken {
				if err := SaveToken(tokenPath, fresh); err != nil {
					return nil, err
				}
			}
			return &persistingSource{base: ts, path: tokenPath, last: fresh.AccessToken}, nil
		}
		var re *oauth2.RetrieveError
		if !errors.As(err, &re) {
			return nil, err
		}
		fmt.Fprintln(out, "保存済みのトークンが無効になりました。再認可します。")
	} else if !errors.Is(err, fs.ErrNotExist) {
		fmt.Fprintf(out, "トークンを読み込めませんでした (%v)。再認可します。\n", err)
	}
	tok, err = Authorize(ctx, cfg, openURL, out)
	if err != nil {
		return nil, err
	}
	if err := SaveToken(tokenPath, tok); err != nil {
		return nil, err
	}
	return &persistingSource{base: cfg.TokenSource(ctx, tok), path: tokenPath, last: tok.AccessToken}, nil
}

type persistingSource struct {
	base oauth2.TokenSource
	path string
	mu   sync.Mutex
	last string
}

func (s *persistingSource) Token() (*oauth2.Token, error) {
	tok, err := s.base.Token()
	if err != nil {
		return nil, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if tok.AccessToken != s.last {
		s.last = tok.AccessToken
		_ = SaveToken(s.path, tok) // a failed write only costs a refresh next launch
	}
	return tok, nil
}

// Package text measures and shapes terminal text. All width math in gsv goes
// through here so that ambiguous-width runes (…, │, ─) always count as 1.
package text

import (
	"strings"

	"github.com/mattn/go-runewidth"
)

var cond = func() *runewidth.Condition {
	c := runewidth.NewCondition()
	c.EastAsianWidth = false
	return c
}()

var flattener = strings.NewReplacer("\r\n", " ", "\n", " ", "\r", " ", "\t", " ")

func Width(s string) int { return cond.StringWidth(s) }

func Flatten(s string) string { return flattener.Replace(s) }

// Fit flattens s, truncates it with "…" and pads it to exactly w columns.
func Fit(s string, w int) string {
	if w <= 0 {
		return ""
	}
	return cond.FillRight(Truncate(Flatten(s), w), w)
}

// Truncate shortens s to at most w columns, ending with "…" when cut.
func Truncate(s string, w int) string {
	if w <= 0 {
		return ""
	}
	if cond.StringWidth(s) <= w {
		return s
	}
	return cond.Truncate(s, w, "…")
}

// Wrap splits s on newlines and then breaks each line at w columns.
func Wrap(s string, w int) []string {
	w = max(w, 1)
	var out []string
	for _, para := range strings.Split(strings.ReplaceAll(s, "\r\n", "\n"), "\n") {
		var line strings.Builder
		lw := 0
		for _, r := range para {
			if r == '\t' {
				r = ' '
			}
			rw := cond.RuneWidth(r)
			if lw+rw > w && lw > 0 {
				out = append(out, line.String())
				line.Reset()
				lw = 0
			}
			line.WriteRune(r)
			lw += rw
		}
		out = append(out, line.String())
	}
	return out
}

// Block renders lines as exactly h rows of exactly w columns.
func Block(lines []string, w, h int) string {
	out := make([]string, max(h, 0))
	for i := range out {
		s := ""
		if i < len(lines) {
			s = lines[i]
		}
		out[i] = Fit(s, w)
	}
	return strings.Join(out, "\n")
}

// Matcher reports whether a string contains q, case-insensitively unless q has an uppercase letter.
func Matcher(q string) func(string) bool {
	if strings.ToLower(q) != q {
		return func(s string) bool { return strings.Contains(s, q) }
	}
	return func(s string) bool { return strings.Contains(strings.ToLower(s), q) }
}

// CellWidth is the fixed display width of every column: one space of padding plus 11 columns of content.
const CellWidth = 12

var (
	cursorFocused = lipgloss.NewStyle().Reverse(true)
	cursorBlurred = lipgloss.NewStyle().Underline(true)

// ColumnName converts a 0-based column index to A, B, …, Z, AA, ….
func ColumnName(i int) string {
	name := ""
	for i++; i > 0; i = (i - 1) / 26 {
		name = string(rune('A'+(i-1)%26)) + name
	}
	return name
}

type geometry struct {
	gutter  int // width of the row-number column
	visCols int
	visRows int
}

// layoutFor computes how many columns and data rows fit in a body of width×height.
func layoutFor(nRows, width, height int) geometry {
	g := geometry{gutter: max(len(strconv.Itoa(nRows)), 2)}
	g.visCols = max((width-g.gutter-1)/(CellWidth+1), 1)
	g.visRows = max((height-2)/2, 1)
	return g
}

func cellAt(rows [][]string, r, c int) string {
	if r < 0 || r >= len(rows) || c < 0 || c >= len(rows[r]) {
		return ""
	}
	return rows[r][c]
}

type frame struct {
	rows                            [][]string
	nCols                           int
	curRow, curCol, topRow, leftCol int
	width, height                   int
	focused                         bool
}

func separator(gutter, n int) string {
	return strings.Repeat("─", gutter) + "┼" +
		strings.Repeat(strings.Repeat("─", CellWidth)+"┼", n-1) +
		strings.Repeat("─", CellWidth) + "┤"
}

// render draws the ruled grid body: a column header, then each row followed by a separator line.
func render(f frame) string {
	if f.height <= 0 {
		return ""
	}
	g := layoutFor(len(f.rows), f.width, f.height)
	if f.width < g.gutter+CellWidth+2 {
		return text.Block([]string{"(幅が狭すぎます)"}, f.width, f.height)
	}
	if len(f.rows) == 0 || f.nCols == 0 {
		return text.Block([]string{"(空のシート)"}, f.width, f.height)
	}
	lastCol := min(f.leftCol+g.visCols, f.nCols)
	cursor := cursorBlurred
	if f.focused {
		cursor = cursorFocused
	}

	var lines []string
	var b strings.Builder
	b.WriteString(strings.Repeat(" ", g.gutter) + "│")
	for c := f.leftCol; c < lastCol; c++ {
		b.WriteString(" " + text.Fit(ColumnName(c), CellWidth-1) + "│")
	}
	sep := separator(g.gutter, lastCol-f.leftCol)
	lines = append(lines, b.String(), sep)

	for r := f.topRow; r < min(f.topRow+g.visRows, len(f.rows)); r++ {
		b.Reset()
		b.WriteString(fmt.Sprintf("%*d│", g.gutter, r+1))
		for c := f.leftCol; c < lastCol; c++ {
			cell := " " + text.Fit(cellAt(f.rows, r, c), CellWidth-1)
			if r == f.curRow && c == f.curCol {
				cell = cursor.Render(cell)
			}
			b.WriteString(cell + "│")
		}
		lines = append(lines, b.String(), sep)
	}
	for len(lines) < f.height {
		lines = append(lines, "")
	}
	return strings.Join(lines[:f.height], "\n")
}

// Package grid holds the right-pane spreadsheet view.
package grid

import (
	"fmt"
	"strconv"
	"strings"

	"gsv/internal/gsheets"
	"gsv/internal/ui/keymap"
	"gsv/internal/ui/text"
)

// Effect tells the caller what to do after an action.
type Effect int

const (
	None     Effect = iota
	LoadTab         // fetch values for CurrentTab()
	Peek            // show the current cell in a popup
	NotFound        // a repeated search found nothing
)

// LargeSheetCells is the cell count above which the caller warns the user.
const LargeSheetCells = 200000

type Model struct {
	id, title   string
	tabs        []gsheets.Tab
	tab         int
	data        map[int][][]string
	nCols       map[int]int
	loading     map[int]bool
	errs        map[int]error
	openLoading bool
	openErr     error

	curRow, curCol, topRow, leftCol int
	width, height                   int

	query   string
	forward bool
}

func New() *Model {
	return &Model{data: map[int][][]string{}, nCols: map[int]int{}, loading: map[int]bool{}, errs: map[int]error{}}
}

func (m *Model) SetSize(w, h int) {
	m.width, m.height = w, h
	m.clamp()
}

// Open resets the grid for spreadsheet id and marks it as loading.
func (m *Model) Open(id string) {
	*m = Model{
		id: id, width: m.width, height: m.height, openLoading: true,
		data: map[int][][]string{}, nCols: map[int]int{}, loading: map[int]bool{}, errs: map[int]error{},
	}
}

func (m *Model) SpreadsheetID() string { return m.id }

func (m *Model) SetTabs(title string, tabs []gsheets.Tab) Effect {
	m.title, m.tabs, m.tab, m.openLoading = title, tabs, 0, false
	if len(tabs) == 0 {
		return None
	}
	m.loading[0] = true
	return LoadTab
}

func (m *Model) SetOpenError(err error) {
	m.openLoading, m.openErr = false, err
}

func (m *Model) CurrentTab() int { return m.tab }

func (m *Model) TabTitle(i int) string {
	if i < 0 || i >= len(m.tabs) {
		return ""
	}
	return m.tabs[i].Title
}

func (m *Model) SetTabData(tab int, rows [][]string) {
	n := 0
	for _, r := range rows {
		n = max(n, len(r))
	}
	m.data[tab], m.nCols[tab] = rows, n
	delete(m.loading, tab)
	delete(m.errs, tab)
	m.clamp()
}

func (m *Model) SetTabError(tab int, err error) {
	delete(m.loading, tab)
	m.errs[tab] = err
}

func (m *Model) rows() [][]string { return m.data[m.tab] }

func (m *Model) geom() geometry { return layoutFor(len(m.rows()), m.width, m.height-1) }

func (m *Model) Apply(a keymap.Action) Effect {
	switch a.Kind {
	case keymap.Down:
		m.curRow += a.Count
	case keymap.Up:
		m.curRow -= a.Count
	case keymap.Right:
		m.curCol += a.Count
	case keymap.Left:
		m.curCol -= a.Count
	case keymap.Top:
		m.curRow = 0
		if a.HasCount {
			m.curRow = a.Count - 1
		}
	case keymap.Bottom:
		m.curRow = len(m.rows()) - 1
		if a.HasCount {
			m.curRow = a.Count - 1
		}
	case keymap.LineStart:
		m.curCol = 0
	case keymap.LineEnd:
		if rows := m.rows(); m.curRow < len(rows) {
			m.curCol = len(rows[m.curRow]) - 1
		}
	case keymap.HalfPageDown:
		m.curRow += max(m.geom().visRows/2, 1)
	case keymap.HalfPageUp:
		m.curRow -= max(m.geom().visRows/2, 1)
	case keymap.NextTab:
		if n := len(m.tabs); n > 0 {
			if a.HasCount {
				if a.Count <= n {
					return m.switchTab(a.Count - 1)
				}
				break
			}
			return m.switchTab((m.tab + 1) % n)
		}
	case keymap.PrevTab:
		if n := len(m.tabs); n > 0 {
			return m.switchTab(((m.tab-a.Count)%n + n) % n)
		}
	case keymap.Open, keymap.Peek:
		if _, _, ok := m.CurrentCell(); ok {
			return Peek
		}
	case keymap.Reload:
		if len(m.tabs) > 0 {
			m.loading[m.tab] = true
			delete(m.errs, m.tab)
			return LoadTab
		}
	case keymap.NextMatch, keymap.PrevMatch:
		if m.query != "" {
			forward := m.forward == (a.Kind == keymap.NextMatch)
			if !m.find(m.query, forward) {
				return NotFound
			}
		}
	}
	m.clamp()
	return None
}

func (m *Model) switchTab(i int) Effect {
	m.tab, m.curRow, m.curCol, m.topRow, m.leftCol = i, 0, 0, 0, 0
	m.clamp()
	if _, ok := m.data[i]; !ok && !m.loading[i] {
		m.loading[i] = true
		delete(m.errs, i)
		return LoadTab
	}
	return None
}

// clamp keeps the cursor inside the data and scrolls it into view.
func (m *Model) clamp() {
	nRows, nCols := len(m.rows()), m.nCols[m.tab]
	m.curRow = min(max(m.curRow, 0), max(nRows-1, 0))
	m.curCol = min(max(m.curCol, 0), max(nCols-1, 0))
	g := m.geom()
	if m.curRow < m.topRow {
		m.topRow = m.curRow
	}
	if m.curRow >= m.topRow+g.visRows {
		m.topRow = m.curRow - g.visRows + 1
	}
	if m.curCol < m.leftCol {
		m.leftCol = m.curCol
	}
	if m.curCol >= m.leftCol+g.visCols {
		m.leftCol = m.curCol - g.visCols + 1
	}
}

// Search remembers q for n/N and jumps to the next matching cell in row-major order.
func (m *Model) Search(q string, forward bool) bool {
	m.query, m.forward = q, forward
	return m.find(q, forward)
}

func (m *Model) Query() string { return m.query }

func (m *Model) find(q string, forward bool) bool {
	rows, nCols := m.rows(), m.nCols[m.tab]
	total := len(rows) * nCols
	if total == 0 {
		return false
	}
	match := text.Matcher(q)
	start := m.curRow*nCols + m.curCol
	for i := 1; i <= total; i++ {
		step := i
		if !forward {
			step = -i
		}
		p := ((start+step)%total + total) % total
		if r, c := p/nCols, p%nCols; match(cellAt(rows, r, c)) {
			m.curRow, m.curCol = r, c
			m.clamp()
			return true
		}
	}
	return false
}

// CurrentCell returns the cursor's A1 address and raw value; ok is false when no data is shown.
func (m *Model) CurrentCell() (addr, value string, ok bool) {
	rows := m.rows()
	if len(rows) == 0 || m.nCols[m.tab] == 0 {
		return "", "", false
	}
	return ColumnName(m.curCol) + strconv.Itoa(m.curRow+1), cellAt(rows, m.curRow, m.curCol), true
}

func (m *Model) Location() string {
	if m.id == "" || m.title == "" {
		return ""
	}
	if t := m.TabTitle(m.tab); t != "" {
		return m.title + " / " + t
	}
	return m.title
}

func (m *Model) titleLine() string {
	if m.id == "" {
		return ""
	}
	parts := []string{m.title}
	for i, t := range m.tabs {
		if i == m.tab {
			parts = append(parts, "["+t.Title+"]")
		} else {
			parts = append(parts, t.Title)
		}
	}
	return strings.Join(parts, "  ")
}

// View renders the title line plus the body, exactly height lines.
func (m *Model) View(focused bool) string {
	if m.height <= 0 {
		return ""
	}
	title := text.Fit(m.titleLine(), m.width)
	bodyH := m.height - 1
	var body string
	switch {
	case m.id == "":
		body = text.Block([]string{"シートが開かれていません。", "左のツリーで l / Enter、または :e <URL>"}, m.width, bodyH)
	case m.openLoading:
		body = text.Block([]string{"⟳ loading…"}, m.width, bodyH)
	case m.openErr != nil:
		body = text.Block([]string{"! 読み込み失敗: " + m.openErr.Error()}, m.width, bodyH)
	case m.errs[m.tab] != nil:
		body = text.Block([]string{fmt.Sprintf("! 読み込み失敗: %v (r で再試行)", m.errs[m.tab])}, m.width, bodyH)
	case m.loading[m.tab] || (len(m.tabs) > 0 && !m.loaded(m.tab)):
		body = text.Block([]string{"⟳ loading…"}, m.width, bodyH)
	default:
		body = render(frame{
			rows: m.rows(), nCols: m.nCols[m.tab],
			curRow: m.curRow, curCol: m.curCol, topRow: m.topRow, leftCol: m.leftCol,
			width: m.width, height: bodyH, focused: focused,
		})
	}
	if bodyH <= 0 {
		return title
	}
	return title + "\n" + body
}

func (m *Model) loaded(tab int) bool {
	_, ok := m.data[tab]
	return ok
}

// Package tree holds the left-pane Drive tree.
package tree

import (
	"strings"

	"github.com/charmbracelet/lipgloss"

	"gsv/internal/config"
	"gsv/internal/gdrive"
	"gsv/internal/ui/keymap"
	"gsv/internal/ui/text"
)

type RootKind int

const (
	NotRoot RootKind = iota
	MyDrive
	SharedWithMe
	Custom
	SearchResults
)

// Node is a tree entry. ID and Root never change after creation, so loaders may read them concurrently.
type Node struct {
	ID       string // Drive file ID; "root" for My Drive; the query for search results
	Name     string
	Kind     gdrive.Kind
	Root     RootKind
	Children []*Node
	Expanded bool
	Loaded   bool
	Loading  bool
	Err      error

	parent *Node
	depth  int
}

func (n *Node) IsSheet() bool { return n.Kind == gdrive.KindSpreadsheet }

type EffectKind int

const (
	None         EffectKind = iota
	LoadChildren            // fetch Node's children
	OpenSheet               // open Node in the grid
	NotFound                // a repeated search found nothing
)

type Effect struct {
	Kind EffectKind
	Node *Node
}

var (
	cursorFocused = lipgloss.NewStyle().Reverse(true)
	cursorBlurred = lipgloss.NewStyle().Bold(true)
)

type Model struct {
	roots         []*Node
	cursor, top   int
	width, height int
	query         string
	forward       bool
}

func New(shared []config.SharedRoot) *Model {
	m := &Model{roots: []*Node{
		{ID: "root", Name: "My Drive", Kind: gdrive.KindFolder, Root: MyDrive},
		{Name: "Shared with me", Kind: gdrive.KindFolder, Root: SharedWithMe},
	}}
	for _, s := range shared {
		m.roots = append(m.roots, &Node{ID: s.ID, Name: s.Name, Kind: gdrive.KindFolder, Root: Custom})
	}
	return m
}

func (m *Model) SetSize(w, h int) {
	m.width, m.height = w, h
	m.clamp()
}

func (m *Model) visible() []*Node {
	var out []*Node
	var walk func(ns []*Node)
	walk = func(ns []*Node) {
		for _, n := range ns {
			out = append(out, n)
			if n.Expanded {
				walk(n.Children)
			}
		}
	}
	walk(m.roots)
	return out
}

func (m *Model) Selected() *Node {
	vis := m.visible()
	if m.cursor < 0 || m.cursor >= len(vis) {
		return nil
	}
	return vis[m.cursor]
}

func (m *Model) indexOf(n *Node) int {
	for i, v := range m.visible() {
		if v == n {
			return i
		}
	}
	return m.cursor
}

func (m *Model) bodyHeight() int { return max(m.height-1, 1) }

func (m *Model) Apply(a keymap.Action) Effect {
	sel := m.Selected()
	switch a.Kind {
	case keymap.Down:
		m.cursor += a.Count
	case keymap.Up:
		m.cursor -= a.Count
	case keymap.Top:
		m.cursor = 0
		if a.HasCount {
			m.cursor = a.Count - 1
		}
	case keymap.Bottom:
		m.cursor = len(m.visible()) - 1
		if a.HasCount {
			m.cursor = a.Count - 1
		}
	case keymap.HalfPageDown:
		m.cursor += max(m.bodyHeight()/2, 1)
	case keymap.HalfPageUp:
		m.cursor -= max(m.bodyHeight()/2, 1)
	case keymap.Right, keymap.Open:
		if sel == nil {
			break
		}
		if sel.IsSheet() {
			return Effect{Kind: OpenSheet, Node: sel}
		}
		return m.expand(sel)
	case keymap.Left:
		if sel == nil {
			break
		}
		if sel.Expanded {
			sel.Expanded = false
		} else if sel.parent != nil {
			m.cursor = m.indexOf(sel.parent)
		}
	case keymap.Reload:
		target := sel
		if sel != nil && sel.IsSheet() {
			target = sel.parent
		}
		if target != nil {
			target.Expanded, target.Loading, target.Err = true, true, nil
			return Effect{Kind: LoadChildren, Node: target}
		}
	case keymap.NextMatch, keymap.PrevMatch:
		if m.query != "" {
			forward := m.forward == (a.Kind == keymap.NextMatch)
			if !m.find(m.query, forward) {
				return Effect{Kind: NotFound}
			}
		}
	}
	m.clamp()
	return Effect{}
}

func (m *Model) expand(n *Node) Effect {
	n.Expanded = true
	if !n.Loaded && !n.Loading {
		n.Loading, n.Err = true, nil
		return Effect{Kind: LoadChildren, Node: n}
	}
	return Effect{}
}

func (m *Model) SetChildren(n *Node, items []gdrive.Item, err error) {
	n.Loading = false
	if err != nil {
		n.Err = err
		m.clamp()
		return
	}
	n.Err, n.Loaded = nil, true
	n.Children = make([]*Node, 0, len(items))
	for _, it := range items {
		n.Children = append(n.Children, &Node{ID: it.ID, Name: it.Name, Kind: it.Kind, parent: n, depth: n.depth + 1})
	}
	m.clamp()
}

// BeginSearch puts a fresh "Search: query" root at the top and selects it.
func (m *Model) BeginSearch(query string) Effect {
	roots := m.roots[:0:0]
	for _, r := range m.roots {
		if r.Root != SearchResults {
			roots = append(roots, r)
		}
	}
	n := &Node{ID: query, Name: "Search: " + query, Kind: gdrive.KindFolder, Root: SearchResults,
		Expanded: true, Loading: true}
	m.roots = append([]*Node{n}, roots...)
	m.cursor = 0
	m.clamp()
	return Effect{Kind: LoadChildren, Node: n}
}

func (m *Model) Search(q string, forward bool) bool {
	m.query, m.forward = q, forward
	return m.find(q, forward)
}

func (m *Model) Query() string { return m.query }

// find searches every loaded node (expanded or not) in tree order and reveals the hit.
func (m *Model) find(q string, forward bool) bool {
	var all []*Node
	var walk func(ns []*Node)
	walk = func(ns []*Node) {
		for _, n := range ns {
			all = append(all, n)
			walk(n.Children)
		}
	}
	walk(m.roots)
	start := 0
	for i, n := range all {
		if n == m.Selected() {
			start = i
		}
	}
	match := text.Matcher(q)
	for i := 1; i <= len(all); i++ {
		step := i
		if !forward {
			step = -i
		}
		n := all[((start+step)%len(all)+len(all))%len(all)]
		if match(n.Name) {
			for p := n.parent; p != nil; p = p.parent {
				p.Expanded = true
			}
			m.cursor = m.indexOf(n)
			m.clamp()
			return true
		}
	}
	return false
}

func (m *Model) clamp() {
	n := len(m.visible())
	m.cursor = min(max(m.cursor, 0), max(n-1, 0))
	h := m.bodyHeight()
	if m.cursor < m.top {
		m.top = m.cursor
	}
	if m.cursor >= m.top+h {
		m.top = m.cursor - h + 1
	}
}

func label(n *Node) string {
	icon := "▤"
	if !n.IsSheet() {
		icon = "▸"
		if n.Expanded {
			icon = "▾"
		}
	}
	s := strings.Repeat("  ", n.depth) + icon + " " + n.Name
	switch {
	case n.Loading:
		s += " ⟳ loading…"
	case n.Err != nil:
		s += " ! 読み込み失敗"
	}
	return s
}

// View renders "Drive" and the visible nodes, exactly height lines of width columns.
func (m *Model) View(focused bool) string {
	if m.height <= 0 || m.width <= 0 {
		return ""
	}
	cursor := cursorBlurred
	if focused {
		cursor = cursorFocused
	}
	lines := []string{text.Fit("Drive", m.width)}
	vis := m.visible()
	for i := m.top; i < min(m.top+m.bodyHeight(), len(vis)); i++ {
		line := text.Fit(label(vis[i]), m.width)
		if i == m.cursor {
			line = cursor.Render(line)
		}
		lines = append(lines, line)
	}
	for len(lines) < m.height {
		lines = append(lines, strings.Repeat(" ", m.width))
	}
	return strings.Join(lines[:m.height], "\n")
}

// Package cmdline implements the bottom-line ":" / "/" / "?" input and command parsing.
package cmdline

import (
	"regexp"
	"strings"

	tea "github.com/charmbracelet/bubbletea"
)

type Mode int

const (
	Command Mode = iota
	SearchForward
	SearchBackward
)

type Result int

const (
	Pending Result = iota
	Submitted
	Cancelled
)

type Input struct {
	active bool
	mode   Mode
	buf    []rune
}

func (in *Input) Start(mode Mode) { in.active, in.mode, in.buf = true, mode, nil }
func (in *Input) Active() bool    { return in.active }
func (in *Input) Mode() Mode      { return in.mode }

// Key edits the line. On Submitted it returns the typed text without the prefix.
func (in *Input) Key(msg tea.KeyMsg) (Result, string) {
	switch msg.Type {
	case tea.KeyEnter:
		in.active = false
		return Submitted, string(in.buf)
	case tea.KeyEsc, tea.KeyCtrlC:
		in.active = false
		return Cancelled, ""
	case tea.KeyBackspace:
		if len(in.buf) == 0 {
			in.active = false
			return Cancelled, ""
		}
		in.buf = in.buf[:len(in.buf)-1]
	case tea.KeyCtrlU:
		in.buf = nil
	case tea.KeySpace:
		in.buf = append(in.buf, ' ')
	case tea.KeyRunes:
		in.buf = append(in.buf, msg.Runes...)
	}
	return Pending, ""
}

func (in *Input) View() string {
	prefix := ":"
	switch in.mode {
	case SearchForward:
		prefix = "/"
	case SearchBackward:
		prefix = "?"
	}
	return prefix + string(in.buf) + "█"
}

// Cmd is a parsed ex command: Name is the first word, Arg the trimmed rest, Raw the whole trimmed line.
type Cmd struct {
	Name, Arg, Raw string
}

func Parse(s string) Cmd {
	raw := strings.TrimSpace(s)
	name, arg, _ := strings.Cut(raw, " ")
	return Cmd{Name: name, Arg: strings.TrimSpace(arg), Raw: raw}
}

var (
	urlID  = regexp.MustCompile(`/spreadsheets/(?:u/\d+/)?d/([A-Za-z0-9_-]+)`)
	bareID = regexp.MustCompile(`^[A-Za-z0-9_-]{20,}$`)
)

// ParseSpreadsheetID accepts a Sheets URL or a bare spreadsheet ID.
func ParseSpreadsheetID(s string) (string, bool) {
	s = strings.TrimSpace(s)
	if m := urlID.FindStringSubmatch(s); m != nil {
		return m[1], true
	}
	if bareID.MatchString(s) {
		return s, true
	}
	return "", false
}

func keyMsg(k string) tea.KeyMsg {
	switch k {
	case "enter":
		return tea.KeyMsg{Type: tea.KeyEnter}

		return tea.KeyMsg{Type: tea.KeyCtrlN}
	case "ctrl+c":
		return tea.KeyMsg{Type: tea.KeyCtrlC}
	}
	return tea.KeyMsg{Type: tea.KeyRunes, Runes: []rune(k)}
}

// drain runs cmd synchronously, feeding results back into m. It reports whether tea.Quit was produced.
func drain(m *Model, cmd tea.Cmd) (quit bool) {
	if cmd == nil {
		return false
	}
	switch msg := cmd().(type) {
	case tea.QuitMsg:
		return true
	case tea.BatchMsg:
		for _, c := range msg {
			quit = drain(m, c) || quit
		}
		return quit
	case nil:
		return false
	default:
		_, next := m.Update(msg)
		return drain(m, next)
	}
}

func press(m *Model, keys ...string) (quit bool) {
	for _, k := range keys {
		_, cmd := m.Update(keyMsg(k))
		quit = drain(m, cmd) || quit
	}
	return quit
}

// typeLine types s (starting with ":" or "/") and presses enter.
func typeLine(m *Model, s string) bool {
	press(m, s[:1])
	_, cmd := m.Update(tea.KeyMsg{Type: tea.KeyRunes, Runes: []rune(s[1:])})
	drain(m, cmd)
	return press(m, "enter")
}

func statusLine(m *Model) string {
	lines := strings.Split(m.View(), "\n")
	return lines[len(lines)-1]
}

package ui

import "gsv/internal/ui/text"

// popup shows one cell's full value, wrapped to the pane width.
type popup struct {
	title string
	lines []string
	off   int
}

func newPopup(addr, value string, width int) *popup {
	return &popup{title: addr, lines: text.Wrap(value, max(width-2, 1))}
}

func (p *popup) scroll(d, height int) {
	p.off = min(max(p.off+d, 0), max(len(p.lines)-(height-1), 0))
}

func (p *popup) View(width, height int) string {
	out := []string{"── " + p.title + " ──  (j/k でスクロール、q / Esc で閉じる)"}
	for i := p.off; i < len(p.lines) && len(out) < height; i++ {
		out = append(out, " "+p.lines[i])
	}
	return text.Block(out, width, height)
}

// Package ui wires the tree, grid and command line into the gsv TUI.
package ui

import (
	"context"
	"fmt"
	"strings"
	"time"

	tea "github.com/charmbracelet/bubbletea"
	"github.com/charmbracelet/lipgloss"

	"gsv/internal/config"
	"gsv/internal/gdrive"
	"gsv/internal/gsheets"
	"gsv/internal/ui/cmdline"
	"gsv/internal/ui/grid"
	"gsv/internal/ui/keymap"
	"gsv/internal/ui/text"
	"gsv/internal/ui/tree"
)

type Deps struct {
	Drive  gdrive.Client
	Sheets gsheets.Client
	Shared []config.SharedRoot
}

type focus int

const (
	focusTree focus = iota
	focusGrid
)

const apiTimeout = 60 * time.Second

var errStyle = lipgloss.NewStyle().Foreground(lipgloss.Color("9"))

type childrenMsg struct {
	node  *tree.Node
	items []gdrive.Item
	err   error
}

type tabsMsg struct {
	id, title string
	tabs      []gsheets.Tab
	err       error
}

type valuesMsg struct {
	id   string
	tab  int
	rows [][]string
	err  error
}

type Model struct {
	deps       Deps
	tree       *tree.Model
	grid       *grid.Model
	keys       keymap.Parser
	input      cmdline.Input
	popup      *popup
	focus      focus
	treeHidden bool
	width      int
	height     int
	msg        string
	msgErr     bool
}

func New(d Deps) *Model {
	return &Model{deps: d, tree: tree.New(d.Shared), grid: grid.New()}
}

// Init expands My Drive, which starts selected.
func (m *Model) Init() tea.Cmd {
	return m.treeEffect(m.tree.Apply(keymap.Action{Kind: keymap.Open, Count: 1}))
}

func (m *Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height
		m.resize()
	case tea.KeyMsg:
		return m, m.handleKey(msg)
	case childrenMsg:
		m.tree.SetChildren(msg.node, msg.items, msg.err)
		if msg.err != nil {
			m.setErr("Drive: " + msg.err.Error())
		}
	case tabsMsg:
		if msg.id != m.grid.SpreadsheetID() {
			break
		}
		if msg.err != nil {
			m.grid.SetOpenError(msg.err)
			m.setErr("Sheets: " + msg.err.Error())
			break
		}
		if m.grid.SetTabs(msg.title, msg.tabs) == grid.LoadTab {
			return m, m.loadValues()
		}
	case valuesMsg:
		if msg.id != m.grid.SpreadsheetID() {
			break
		}
		if msg.err != nil {
			m.grid.SetTabError(msg.tab, msg.err)
			m.setErr("Sheets: " + msg.err.Error())
			break
		}
		m.grid.SetTabData(msg.tab, msg.rows)
		if n := len(msg.rows) * maxLen(msg.rows); n > grid.LargeSheetCells {
			m.msg, m.msgErr = fmt.Sprintf("警告: %d セルの大きなシートです（表示が重くなる場合があります）", n), false
		}
	}
	return m, nil
}

func maxLen(rows [][]string) int {
	n := 0
	for _, r := range rows {
		n = max(n, len(r))
	}
	return n
}

func (m *Model) setErr(s string) { m.msg, m.msgErr = s, true }

func (m *Model) resize() {
	treeW := 0
	if !m.treeHidden {
		treeW = 30
		if m.width < 80 {
			treeW = m.width / 3
		}
	}
	gridW := m.width
	if !m.treeHidden {
		gridW = m.width - treeW - 1
	}
	paneH := max(m.height-1, 0)
	m.tree.SetSize(treeW, paneH)
	m.grid.SetSize(gridW, paneH)
}

func (m *Model) handleKey(msg tea.KeyMsg) tea.Cmd {
	if m.popup != nil {
		switch msg.String() {
		case "q", "esc", "enter", "K":
			m.popup = nil
		case "j", "down":
			m.popup.scroll(1, m.height-1)
		case "k", "up":
			m.popup.scroll(-1, m.height-1)
		case "ctrl+c":
			return tea.Quit
		}
		return nil
	}
	if m.input.Active() {
		mode := m.input.Mode()
		res, s := m.input.Key(msg)
		if res != cmdline.Submitted {
			return nil
		}
		return m.submit(mode, s)
	}
	if msg.String() == "ctrl+c" {
		return tea.Quit
	}
	m.msg = ""
	a, ok := m.keys.Feed(msg.String())
	if !ok {
		return nil
	}
	switch a.Kind {
	case keymap.CommandMode:
		m.input.Start(cmdline.Command)
	case keymap.SearchForward:
		m.input.Start(cmdline.SearchForward)
	case keymap.SearchBackward:
		m.input.Start(cmdline.SearchBackward)
	case keymap.PaneLeft:
		if !m.treeHidden {
			m.focus = focusTree
		}
	case keymap.PaneRight:
		m.focus = focusGrid
	case keymap.PaneNext:
		if !m.treeHidden {
			m.focus = 1 - m.focus
		}
	case keymap.ToggleTree:
		m.treeHidden = !m.treeHidden
		if m.treeHidden {
			m.focus = focusGrid
		}
		m.resize()
	case keymap.Cancel:
	default:
		if m.focus == focusTree {
			return m.treeEffect(m.tree.Apply(a))
		}
		return m.gridEffect(m.grid.Apply(a))
	}
	return nil
}

func (m *Model) submit(mode cmdline.Mode, s string) tea.Cmd {
	if mode == cmdline.Command {
		return m.execute(cmdline.Parse(s))
	}
	forward := mode == cmdline.SearchForward
	var found bool
	if m.focus == focusTree {
		if s == "" {
			s = m.tree.Query()
		}
		found = s != "" && m.tree.Search(s, forward)
	} else {
		if s == "" {
			s = m.grid.Query()
		}
		found = s != "" && m.grid.Search(s, forward)
	}
	if !found {
		m.setErr("E486: Pattern not found: " + s)
	}
	return nil
}

func (m *Model) execute(c cmdline.Cmd) tea.Cmd {
	switch c.Name {
	case "":
		return nil
	case "q", "q!", "qa", "qa!", "quit", "qall":
		return tea.Quit
	case "e", "edit":
		if c.Arg == "" {
			m.setErr("E471: Argument required")
			return nil
		}
		id, ok := cmdline.ParseSpreadsheetID(c.Arg)
		if !ok {
			m.setErr("スプレッドシートの URL / ID ではありません: " + c.Arg)
			return nil
		}
		return m.open(id)
	case "find":
		if c.Arg == "" {
			m.setErr("E471: Argument required")
			return nil
		}
		if m.treeHidden {
			m.treeHidden = false
			m.resize()
		}
		m.focus = focusTree
		return m.treeEffect(m.tree.BeginSearch(c.Arg))
	}
	m.setErr("E492: Not an editor command: " + c.Raw)
	return nil
}

func (m *Model) treeEffect(e tree.Effect) tea.Cmd {
	switch e.Kind {
	case tree.LoadChildren:
		return m.loadChildren(e.Node)
	case tree.OpenSheet:
		return m.open(e.Node.ID)
	case tree.NotFound:
		m.setErr("E486: Pattern not found: " + m.tree.Query())
	}
	return nil
}

func (m *Model) gridEffect(e grid.Effect) tea.Cmd {
	switch e {
	case grid.LoadTab:
		return m.loadValues()
	case grid.Peek:
		if addr, v, ok := m.grid.CurrentCell(); ok {
			m.popup = newPopup(addr, v, m.width)
		}
	case grid.NotFound:
		m.setErr("E486: Pattern not found: " + m.grid.Query())
	}
	return nil
}

func (m *Model) open(id string) tea.Cmd {
	m.grid.Open(id)
	m.focus = focusGrid
	sheets := m.deps.Sheets
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(context.Background(), apiTimeout)
		defer cancel()
		title, tabs, err := sheets.GetTabs(ctx, id)
		return tabsMsg{id: id, title: title, tabs: tabs, err: err}
	}
}

func (m *Model) loadValues() tea.Cmd {
	id, tab := m.grid.SpreadsheetID(), m.grid.CurrentTab()
	title := m.grid.TabTitle(tab)
	sheets := m.deps.Sheets
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(context.Background(), apiTimeout)
		defer cancel()
		rows, err := sheets.GetValues(ctx, id, title)
		return valuesMsg{id: id, tab: tab, rows: rows, err: err}
	}
}

func (m *Model) loadChildren(n *tree.Node) tea.Cmd {
	drive := m.deps.Drive
	root, id := n.Root, n.ID
	return func() tea.Msg {
		ctx, cancel := context.WithTimeout(context.Background(), apiTimeout)
		defer cancel()
		var items []gdrive.Item
		var err error
		switch root {
		case tree.SharedWithMe:
			items, err = drive.ListSharedWithMe(ctx)
		case tree.SearchResults:
			items, err = drive.SearchSpreadsheets(ctx, id)
		default:
			items, err = drive.ListChildren(ctx, id)
		}
		return childrenMsg{node: n, items: items, err: err}
	}
}

func (m *Model) View() string {
	if m.width <= 0 || m.height <= 0 {
		return ""
	}
	paneH := m.height - 1
	var main string
	switch {
	case m.popup != nil:
		main = m.popup.View(m.width, paneH)
	case m.treeHidden:
		main = m.grid.View(m.focus == focusGrid)
	default:
		divider := strings.TrimSuffix(strings.Repeat("│\n", paneH), "\n")
		main = lipgloss.JoinHorizontal(lipgloss.Top,
			m.tree.View(m.focus == focusTree), divider, m.grid.View(m.focus == focusGrid))
	}
	if paneH <= 0 {
		return m.statusLine()
	}
	return main + "\n" + m.statusLine()
}

func (m *Model) statusLine() string {
	if m.input.Active() {
		return text.Truncate(m.input.View(), m.width)
	}
	if m.msg != "" {
		s := text.Truncate(m.msg, m.width)
		if m.msgErr {
			return errStyle.Render(s)
		}
		return s
	}
	parts := []string{"NORMAL"}
	if loc := m.grid.Location(); loc != "" {
		parts = append(parts, loc)
	}
	if addr, v, ok := m.grid.CurrentCell(); ok {
		parts = append(parts, addr, text.Flatten(v))
	}
	if p := m.keys.Pending(); p != "" {
		parts = append(parts, p)
	}
	return text.Truncate(strings.Join(parts, "  "), m.width)
}

```go
// Command gsv is a read-only, vim-style TUI viewer for Google Sheets.
package main

import (
	"context"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"

	tea "github.com/charmbracelet/bubbletea"
	"golang.org/x/oauth2"
	"google.golang.org/api/drive/v3"
	"google.golang.org/api/option"
	"google.golang.org/api/sheets/v4"

	"gsv/internal/auth"
	"gsv/internal/config"
	"gsv/internal/gdrive"
	"gsv/internal/gsheets"
	"gsv/internal/ui"
)

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, "gsv:", err)
		os.Exit(1)
	}
}

func run() error {
	dir, err := config.Dir()
	if err != nil {
		return err
	}
	cfg, err := config.Load(filepath.Join(dir, "config.yaml"))
	if err != nil {
		return err
	}
	credPath := filepath.Join(dir, "credentials.json")
	oc, err := auth.LoadOAuthConfig(credPath)
	if errors.Is(err, auth.ErrNoCredentials) {
		return fmt.Errorf("%w\nGoogle Cloud Console で作成した OAuth クライアント（デスクトップアプリ）の JSON を\n  %s\nに置いてください。手順は docs/setup.md を参照してください", err, credPath)
	}
	if err != nil {
		return err
	}

	ctx := context.Background()
	ts, err := auth.TokenSource(ctx, oc, filepath.Join(dir, "token.json"), openBrowser, os.Stderr)
	if err != nil {
		return err
	}
	httpClient := oauth2.NewClient(ctx, ts)
	dsvc, err := drive.NewService(ctx, option.WithHTTPClient(httpClient))
	if err != nil {
		return err
	}
	ssvc, err := sheets.NewService(ctx, option.WithHTTPClient(httpClient))
	if err != nil {
		return err
	}

	m := ui.New(ui.Deps{Drive: gdrive.New(dsvc), Sheets: gsheets.New(ssvc), Shared: cfg.SharedRoots})
	_, err = tea.NewProgram(m, tea.WithAltScreen()).Run()
	return err
}

func openBrowser(u string) error { return exec.Command("open", u).Start() }
```

### リフレッシュトークンの有効期限に注意
「外部」かつ公開ステータス「テスト中」のままだと、リフレッシュトークンが 7 日で失効し、週 1 回ブラウザでの再認可が必要になります（gsv は自動で再認可フローを開始します）。
避けたい場合は OAuth 同意画面の公開ステータスを「本番環境」にしてください。自分だけで使う分には「未検証のアプリ」警告を「詳細」→「移動」で進めれば利用できます。
